ec_update_corruption_effect = {

			value = {
				value = "estate(estate_type:nobles_estate).estate_tax_rate"

					value = modifier:ec_global_acceptable_max_tax

					max = 0

			subtract = 0.1
		}

	add_country_modifier = { modifier = ec_nobles_corruption_impact months = 2 mode = replace size = 0 }
